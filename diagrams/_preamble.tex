% ---- node text reads right-to-left, as in the book body -------------------
% The picture sits inside \babelsublr so that its GEOMETRY is left-to-right.
% That also makes the BASE direction of every node's text left-to-right: a
% Persian sentence that contains a Latin word ("ذخیره‌ساز (S1)") then comes out
% with its parts in the wrong order (checked with character coordinates,
% tools/fig_words.py). Setting babel's text and paragraph direction to RTL at
% the start of each node restores the order of the book body. Pure Latin or
% math labels are unaffected.
\makeatletter
\newcommand{\NodeRTL}{\bbl@pardir\@ne\bbl@textdir\@ne}
\makeatother
\tikzset{every node/.append style={execute at begin node=\NodeRTL}}
